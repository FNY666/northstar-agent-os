    24	// 160KB limit for the total size of string content in a log entry.
    25	// The total log entry size limit is 256KB. We leave ~96KB (approx 37%) for JSON overhead (escaping, structure) and other fields.
    26	const GLOBAL_TEXT_LIMIT = 160 * 1024;
    27	
    28	interface StringReference {
    29	  get: () => string | undefined;
    30	  set: (val: string) => void;
    31	  len: () => number;
    32	}
    33	
    34	function getStringReferences(parts: AnyPart[]): StringReference[] {
    35	  const refs: StringReference[] = [];
    36	  for (const part of parts) {
    37	    if (part instanceof TextPart) {
    38	      refs.push({
    39	        get: () => part.content,
    40	        set: (val: string) => (part.content = val),
    41	        len: () => part.content.length,
    42	      });
    43	    } else if (part instanceof ReasoningPart) {
    44	      refs.push({
    45	        get: () => part.content,
    46	        set: (val: string) => (part.content = val),
    47	        len: () => part.content.length,
    48	      });
    49	    } else if (part instanceof ToolCallRequestPart) {
    50	      if (part.arguments) {
    51	        refs.push({
    52	          get: () => part.arguments,
    53	          set: (val: string) => (part.arguments = val),
    54	          len: () => part.arguments?.length ?? 0,
    55	        });
    56	      }
    57	    } else if (part instanceof ToolCallResponsePart) {
    58	      if (part.response) {
    59	        refs.push({
    60	          get: () => part.response,
    61	          set: (val: string) => (part.response = val),
    62	          len: () => part.response?.length ?? 0,
    63	        });
    64	      }
    65	    } else if (part instanceof GenericPart) {
    66	      // eslint-disable-next-line no-restricted-syntax
    67	      if (part.type === 'executableCode' && typeof part['code'] === 'string') {
    68	        refs.push({
    69	          // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
    70	          get: () => part['code'] as string,
    71	          set: (val: string) => (part['code'] = val),
    72	          // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
    73	          len: () => (part['code'] as string).length,
    74	        });
    75	      } else if (
    76	        part.type === 'codeExecutionResult' &&
    77	        // eslint-disable-next-line no-restricted-syntax
    78	        typeof part['output'] === 'string'
    79	      ) {
    80	        refs.push({
    81	          // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
    82	          get: () => part['output'] as string,
    83	          set: (val: string) => (part['output'] = val),
    84	          // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
    85	          len: () => (part['output'] as string).length,
    86	        });
    87	      }
    88	    }
    89	  }
    90	  return refs;
    91	}
    92	
    93	function limitTotalLength(parts: AnyPart[]): void {
    94	  const refs = getStringReferences(parts);
    95	  const totalLength = refs.reduce((sum, ref) => sum + ref.len(), 0);
    96	
    97	  if (totalLength <= GLOBAL_TEXT_LIMIT) {
    98	    return;
    99	  }
   100	
   101	  // Calculate the average budget per part for "large" parts.
   102	  // We identify parts that are larger than the fair share (average) and truncate them.
   103	  const averageSize = GLOBAL_TEXT_LIMIT / refs.length;
   104	
   105	  // Filter out parts that are already small enough to not need truncation
   106	  const largeRefs = refs.filter((ref) => ref.len() > averageSize);
   107	  const smallRefsLength = refs
   108	    .filter((ref) => ref.len() <= averageSize)
   109	    .reduce((sum, ref) => sum + ref.len(), 0);
   110	
   111	  // Distribute the remaining budget among large parts
   112	  const remainingBudget = GLOBAL_TEXT_LIMIT - smallRefsLength;
   113	  const budgetPerLargePart = Math.max(
   114	    1,
   115	    Math.floor(remainingBudget / largeRefs.length),
   116	  );
   117	
   118	  for (const ref of largeRefs) {
   119	    const original = ref.get();
   120	    if (original) {
   121	      ref.set(truncateString(original, budgetPerLargePart));
   122	    }
   123	  }
   124	}

   126	export function toInputMessages(contents: Content[]): InputMessages {
   127	  const messages: ChatMessage[] = [];
   128	  for (const content of contents) {
   129	    messages.push(toChatMessage(content));
   130	  }
   131	  const allParts = messages.flatMap((m) => m.parts);
   132	  limitTotalLength(allParts);
   133	  return messages;
   134	}

   136	function isPart(value: unknown): value is Part {
   137	  return (
   138	    typeof value === 'object' &&
   139	    value !== null &&
   140	    !Array.isArray(value) &&
   141	    !('parts' in value)
   142	  );
   143	}

   145	function toPart(part: PartUnion): Part {
   146	  if (typeof part === 'string') {
   147	    return { text: part };
   148	  }
   149	  return part;
   150	}

   152	function toContent(content: ContentUnion): Content | undefined {
   153	  if (typeof content === 'string') {
   154	    // 1. It's a string
   155	    return {
   156	      parts: [toPart(content)],
   157	    };
   158	  } else if (Array.isArray(content)) {
   159	    // 2. It's an array of parts (PartUnion[])
   160	    return {
   161	      parts: content.map(toPart),
   162	    };
   163	  } else if ('parts' in content) {
   164	    // 3. It's a Content object
   165	    return content;
   166	  } else if (isPart(content)) {
   167	    // 4. It's a single Part object (asserted with type guard)
   168	    return {
   169	      parts: [content],
   170	    };
   171	  } else {
   172	    // 5. Handle any other unexpected case
   173	    return undefined;
   174	  }
   175	}

   231	export function toChatMessage(content?: Content): ChatMessage {
   232	  const message: ChatMessage = {
   233	    role: undefined,
   234	    parts: [],
   235	  };
   236	  if (content && content.parts) {

   245	function toOTelPart(part: Part): AnyPart {
   246	  if (part.thought) {
   247	    if (part.text) {

    21	class FileExporter {
    22	  protected writeStream: fs.WriteStream;
    23	
    24	  constructor(filePath: string) {
    25	    this.writeStream = fs.createWriteStream(filePath, { flags: 'a' });
    26	  }
    27	
    28	  protected serialize(data: unknown): string {
    29	    return safeJsonStringify(data, 2) + '\n';
    30	  }
    31	
    32	  /**
    33	   * Ensures that all pending writes are flushed to the underlying stream.
    34	   */
    35	  forceFlush(): Promise<void> {
    36	    return new Promise((resolve, reject) => {
    37	      if (!this.writeStream.writable) {
    38	        resolve();
    39	        return;
    40	      }
    41	      // write('') will be queued after all previous writes and its callback
    42	      // will be called when it (and thus all previous writes) are flushed.
    43	      this.writeStream.write('', (err) => {
    44	        if (err) {
    45	          reject(err);
    46	        } else {
    47	          resolve();
    48	        }
    49	      });
    50	    });
    51	  }
    52	
    53	  shutdown(): Promise<void> {
    54	    return new Promise((resolve) => {
    55	      this.writeStream.end(resolve);
    56	    });
    57	  }
    58	}

    75	/**
    76	 * Deeply truncates strings in an object to prevent GCP log size limit errors.
    77	 */
    78	function truncateLogPayload(payload: unknown, limit = 200000): unknown {
    79	  if (typeof payload === 'string') {
    80	    return payload.length > limit
    81	      ? payload.substring(0, limit) + '... (truncated due to size)'
    82	      : payload;
    83	  }
    84	  if (Array.isArray(payload)) {
    85	    return payload.map((item) => truncateLogPayload(item, limit));
    86	  }
    87	  if (payload !== null && typeof payload === 'object') {
    88	    const truncatedObj: Record<string, unknown> = {};
    89	    for (const [key, value] of Object.entries(payload)) {
    90	      // Keys are also strings, but usually small. Truncate values.
    91	      truncatedObj[key] = truncateLogPayload(value, limit);
    92	    }
    93	    return truncatedObj;
    94	  }
    95	  return payload;
    96	}

    98	/**
    99	 * Google Cloud Logging exporter that uses the Cloud Logging client
   100	 */
   101	export class GcpLogExporter implements LogRecordExporter {
   102	  private logging: Logging;
   103	  private log: Log;
   104	  private pendingWrites: Array<Promise<void>> = [];
   105	
   106	  constructor(projectId?: string, credentials?: JWTInput) {
   107	    this.logging = new Logging({ projectId, credentials });
   108	    this.log = this.logging.log('gemini_cli');
   109	  }
   110	
   111	  export(
   112	    logs: ReadableLogRecord[],
   113	    resultCallback: (result: ExportResult) => void,
   114	  ): void {
   115	    try {
   116	      const entries = logs.map((log) => {
   117	        const rawPayload = {
   118	          ...log.attributes,
   119	          ...log.resource?.attributes,
   120	          message: log.body,
   121	        };
   122	
   123	        const isStrictTelemetry =
   124	          process.env['GEMINI_STRICT_TELEMETRY_LIMITS'] === 'true';
   125	
   126	        let finalPayload: unknown = rawPayload;
   127	
   128	        if (isStrictTelemetry) {
   129	          // Enforce a strict cap on the entire payload to avoid 256KB limit crashes in CI.
   130	          let safePayload = truncateLogPayload(rawPayload, 10000);
   131	          let payloadString = JSON.stringify(safePayload);
   132	
   133	          if (payloadString && payloadString.length > 100000) {
   134	            // If still too large, apply a stricter limit
   135	            safePayload = truncateLogPayload(rawPayload, 2000);
   136	            payloadString = JSON.stringify(safePayload);
   137	
   138	            if (payloadString && payloadString.length > 100000) {
   139	              safePayload = truncateLogPayload(rawPayload, 5000);
   140	              payloadString = JSON.stringify(safePayload);
   141	
   142	              if (payloadString && payloadString.length > 100000) {
   143	                // Fallback: strip structure and send a truncated raw string
   144	                safePayload = {
   145	                  _warning: 'Payload heavily truncated due to strict limits',
   146	                  data: payloadString.substring(0, 50000) + '... (truncated)',
   147	                };
   148	              }
   149	            }
   150	          }
   151	          finalPayload = safePayload;
   152	        }
   153	
   154	        const entry = this.log.entry(
   155	          {
   156	            severity: this.mapSeverityToCloudLogging(log.severityNumber),
   157	            timestamp: new Date(hrTimeToMilliseconds(log.hrTime)),
   158	            resource: {
   159	              type: 'global',
   160	              labels: {
   161	                project_id: this.logging.projectId,
   162	              },
   163	            },
   164	          },
   165	          // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
   166	          finalPayload as Record<string, unknown>,
   167	        );
   168	        return entry;
   169	      });
   170	
   171	      const writePromise = this.log
   172	        .write(entries)
   173	        .then(() => {
   174	          resultCallback({ code: ExportResultCode.SUCCESS });
   175	        })
   176	        .catch((error: Error) => {
   177	          resultCallback({
   178	            code: ExportResultCode.FAILED,
   179	            error,
   180	          });
   181	        })
   182	        .finally(() => {
   183	          const index = this.pendingWrites.indexOf(writePromise);
   184	          if (index > -1) {
   185	            // eslint-disable-next-line @typescript-eslint/no-floating-promises
   186	            this.pendingWrites.splice(index, 1);
   187	          }
   188	        });
   189	      this.pendingWrites.push(writePromise);
   190	    } catch (error) {
   191	      resultCallback({
   192	        code: ExportResultCode.FAILED,
   193	        // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
   194	        error: error as Error,
   195	      });
   196	    }
   197	  }
   198	
   199	  async forceFlush(): Promise<void> {
   200	    if (this.pendingWrites.length > 0) {
   201	      await Promise.all(this.pendingWrites);
   202	    }
   203	  }
   204	
   205	  async shutdown(): Promise<void> {
   206	    await this.forceFlush();
   207	    this.pendingWrites = [];
   208	  }
   209	
   210	  private mapSeverityToCloudLogging(severityNumber?: number): string {
   211	    if (!severityNumber) return 'DEFAULT';
   212	
   213	    // Map OpenTelemetry severity numbers to Cloud Logging severity levels
   214	    // https://opentelemetry.io/docs/specs/otel/logs/data-model/#field-severitynumber
   215	    if (severityNumber >= 21) return 'CRITICAL';
   216	    if (severityNumber >= 17) return 'ERROR';
   217	    if (severityNumber >= 13) return 'WARNING';
   218	    if (severityNumber >= 9) return 'INFO';
   219	    if (severityNumber >= 5) return 'DEBUG';
   220	    return 'DEFAULT';
   221	  }
   222	}
