    40	export interface TranslationState {
    41	  streamId: string;
    42	  streamStartEmitted: boolean;
    43	  model: string | undefined;
    44	  eventCounter: number;
    45	  /** Tracks callId → tool name from requests so responses can reference the name. */
    46	  pendingToolNames: Map<string, string>;
    47	}
    48	
    49	export function createTranslationState(streamId?: string): TranslationState {
    50	  return {
    51	    streamId: streamId ?? crypto.randomUUID(),
    52	    streamStartEmitted: false,
    53	    model: undefined,
    54	    eventCounter: 0,
    55	    pendingToolNames: new Map(),
    56	  };
    57	}

    59	// ---------------------------------------------------------------------------

    63	function makeEvent<T extends AgentEventType>(
    64	  type: T,
    65	  state: TranslationState,
    66	  payload: Partial<AgentEvent<T>>,
    67	): AgentEvent {
    68	  const id = `${state.streamId}-${state.eventCounter++}`;
    69	  // TypeScript cannot preserve the specific discriminated union member across
    70	  // this generic object assembly, so keep the narrowing local to the event
    71	  // constructor boundary.
    72	  // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
    73	  return {
    74	    ...payload,
    75	    id,
    76	    timestamp: new Date().toISOString(),
    77	    streamId: state.streamId,
    78	    type,
    79	  } as AgentEvent;
    80	}
    81	
    82	function ensureStreamStart(state: TranslationState, out: AgentEvent[]): void {
    83	  if (!state.streamStartEmitted) {
    84	    out.push(makeEvent('agent_start', state, {}));
    85	    state.streamStartEmitted = true;
    86	  }
    87	}

    93	/**
    94	 * Translates a single ServerGeminiStreamEvent into zero or more AgentEvents.
    95	 * Mutates `state` (counter, flags) as a side effect.
    96	 */
    97	export function translateEvent(
    98	  event: ServerGeminiStreamEvent,
    99	  state: TranslationState,
   100	): AgentEvent[] {
   101	  const out: AgentEvent[] = [];
   102	
   103	  switch (event.type) {
   104	    case GeminiEventType.ModelInfo:
   105	      state.model = event.value;
   106	      ensureStreamStart(state, out);
   107	      out.push(makeEvent('session_update', state, { model: event.value }));
   108	      break;
   109	
   110	    case GeminiEventType.Content:
   111	      ensureStreamStart(state, out);
   112	      out.push(
   113	        makeEvent('message', state, {
   114	          role: 'agent',
   115	          content: [{ type: 'text', text: event.value }],
   116	        }),
   117	      );
   118	      break;
   119	
   120	    case GeminiEventType.Thought:
   121	      ensureStreamStart(state, out);
   122	      out.push(
   123	        makeEvent('message', state, {
   124	          role: 'agent',
   125	          content: [{ type: 'thought', thought: event.value.description }],
   126	          _meta: event.value.subject
   127	            ? { source: 'agent', subject: event.value.subject }
   128	            : { source: 'agent' },
   129	        }),
   130	      );
   131	      break;
   132	
   133	    case GeminiEventType.Citation:
   134	      ensureStreamStart(state, out);
   135	      out.push(
   136	        makeEvent('message', state, {
   137	          role: 'agent',
   138	          content: [{ type: 'text', text: event.value }],
   139	          _meta: { source: 'agent', citation: true },
   140	        }),
   141	      );
   142	      break;
   143	
   144	    case GeminiEventType.Finished:
   145	      handleFinished(event.value, state, out);
   146	      break;
   147	
   148	    case GeminiEventType.Error:
   149	      handleError(event.value.error, state, out);
   150	      break;
   151	
   152	    case GeminiEventType.UserCancelled:
   153	      ensureStreamStart(state, out);
   154	      out.push(
   155	        makeEvent('agent_end', state, {
   156	          reason: 'aborted',
   157	        }),
   158	      );
   159	      break;
   160	
   161	    case GeminiEventType.MaxSessionTurns:
   162	      ensureStreamStart(state, out);
   163	      out.push(
   164	        makeEvent('agent_end', state, {
   165	          reason: 'max_turns',
   166	          data: {
   167	            code: 'MAX_TURNS_EXCEEDED',
   168	          },
   169	        }),
   170	      );
   171	      break;
   172	
   173	    case GeminiEventType.LoopDetected:
   174	      ensureStreamStart(state, out);
   175	      out.push(
   176	        makeEvent('error', state, {
   177	          status: 'INTERNAL',
   178	          message: 'Loop detected, stopping execution',
   179	          fatal: false,
   180	          _meta: { code: 'LOOP_DETECTED' },
   181	        }),
   182	      );
   183	      break;
   184	
   185	    case GeminiEventType.ContextWindowWillOverflow:
   186	      ensureStreamStart(state, out);
   187	      out.push(
   188	        makeEvent('error', state, {
   189	          status: 'RESOURCE_EXHAUSTED',
   190	          message: `Context window will overflow (estimated: ${event.value.estimatedRequestTokenCount}, remaining: ${event.value.remainingTokenCount})`,
   191	          fatal: true,
   192	        }),
   193	      );
   194	      break;
   195	
   196	    case GeminiEventType.AgentExecutionStopped:
   197	      ensureStreamStart(state, out);
   198	      out.push(
   199	        makeEvent('agent_end', state, {
   200	          reason: 'completed',
   201	          data: {
   202	            message: event.value.systemMessage?.trim() || event.value.reason,
   203	          },
   204	        }),
   205	      );
   206	      break;
   207	
   208	    case GeminiEventType.AgentExecutionBlocked:
   209	      ensureStreamStart(state, out);
   210	      out.push(
   211	        makeEvent('error', state, {
   212	          status: 'PERMISSION_DENIED',
   213	          message: event.value.systemMessage?.trim() || event.value.reason,
   214	          fatal: false,
   215	          _meta: { code: 'AGENT_EXECUTION_BLOCKED' },
   216	        }),
   217	      );
   218	      break;
   219	
   220	    case GeminiEventType.InvalidStream:
   221	      ensureStreamStart(state, out);
   222	      out.push(
   223	        makeEvent('error', state, {
   224	          status: 'INTERNAL',
   225	          message:
   226	            event.value?.message?.trim() ||
   227	            'Invalid stream received from model',
   228	          fatal: true,
   229	          _meta: {
   230	            code: 'INVALID_STREAM',
   231	            errorType: event.value?.type,
   232	            rawMessage: event.value?.message,
   233	          },
   234	        }),
   235	      );
   236	      break;
   237	
   238	    case GeminiEventType.ToolCallRequest:
   239	      ensureStreamStart(state, out);
   240	      state.pendingToolNames.set(event.value.callId, event.value.name);
   241	      out.push(
   242	        makeEvent('tool_request', state, {
   243	          requestId: event.value.callId,
   244	          name: event.value.name,
   245	          args: event.value.args,
   246	          display: event.value.display,
   247	        }),
   248	      );
   249	      break;
   250	
   251	    case GeminiEventType.ToolCallResponse: {
   252	      ensureStreamStart(state, out);
   253	      const data = buildToolResponseData(event.value);
   254	      const display: ToolDisplay | undefined =
   255	        event.value.display ??
   256	        (event.value.resultDisplay
   257	          ? {
   258	              result: toolResultDisplayToDisplayContent(
   259	                event.value.resultDisplay,
   260	              ),
   261	            }
   262	          : undefined);
   263	      out.push(
   264	        makeEvent('tool_response', state, {
   265	          requestId: event.value.callId,
   266	          name: state.pendingToolNames.get(event.value.callId) ?? 'unknown',
   267	          content: event.value.error
   268	            ? [{ type: 'text', text: event.value.error.message }]
   269	            : geminiPartsToContentParts(event.value.responseParts),
   270	          isError: event.value.error !== undefined,
   271	          ...(display ? { display } : {}),
   272	          ...(data ? { data } : {}),
   273	        }),
   274	      );
   275	      state.pendingToolNames.delete(event.value.callId);
   276	      break;
   277	    }
   278	
   279	    case GeminiEventType.ToolCallConfirmation:
   280	      // Elicitations are handled separately by the session layer
   281	      break;
   282	
   283	    // Internal concerns — no AgentEvent emitted
   284	    case GeminiEventType.ChatCompressed:
   285	    case GeminiEventType.Retry:
   286	      break;
   287	
   288	    default:
   289	      ((x: never) => {
   290	        throw new Error(`Unhandled event type: ${JSON.stringify(x)}`);
   291	      })(event);
   292	  }
   293	
   294	  return out;
   295	}

   259	type EventBacklogItem = {
   260	  [K in keyof CoreEvents]: {
   261	    event: K;
   262	    args: CoreEvents[K];
   263	  };
   264	}[keyof CoreEvents];
   265	
   266	export class CoreEventEmitter extends EventEmitter<CoreEvents> {
   267	  private _eventBacklog: EventBacklogItem[] = [];
   268	  private _backlogHead = 0;
   269	  private static readonly MAX_BACKLOG_SIZE = 10000;
   270	
   271	  constructor() {
   272	    super();
   273	  }
   274	
   275	  private _emitOrQueue<K extends keyof CoreEvents>(
   276	    event: K,
   277	    ...args: CoreEvents[K]
   278	  ): void {
   279	    if (this.listenerCount(event) === 0) {
   280	      const backlogSize = this._eventBacklog.length - this._backlogHead;
   281	      if (backlogSize >= CoreEventEmitter.MAX_BACKLOG_SIZE) {
   282	        // Evict oldest entry. Use a head pointer instead of shift() to avoid
   283	        // O(n) array reindexing on every eviction at capacity.
   284	        (this._eventBacklog as unknown[])[this._backlogHead] = undefined;
   285	        this._backlogHead++;
   286	        // Compact once dead entries exceed half capacity to bound memory
   287	        if (this._backlogHead >= CoreEventEmitter.MAX_BACKLOG_SIZE / 2) {
   288	          this._eventBacklog = this._eventBacklog.slice(this._backlogHead);
   289	          this._backlogHead = 0;
   290	        }
   291	      }
   292	      // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
   293	      this._eventBacklog.push({ event, args } as EventBacklogItem);
   294	    } else {
   295	      // eslint-disable-next-line @typescript-eslint/no-unsafe-type-assertion
   296	      (this.emit as (event: K, ...args: CoreEvents[K]) => boolean)(
   297	        event,
   298	        ...args,
   299	      );
   300	    }

   175	          function_name: event.function_name,

   117	export function bufferTelemetryEvent(fn: () => void | Promise<void>): void {
   118	  if (telemetryInitialized) {
   119	    // eslint-disable-next-line @typescript-eslint/no-floating-promises
   120	    fn();
   121	  } else {
   122	    telemetryBuffer.push(fn);
   123	  }
   124	}

   126	async function flushTelemetryBuffer(): Promise<void> {
   127	  if (!telemetryInitialized) return;
   128	  while (telemetryBuffer.length > 0) {
   129	    const fn = telemetryBuffer.shift();
   130	    if (fn) {
   131	      try {
   132	        await fn();
   133	      } catch (e) {
   134	        debugLogger.error('Error executing buffered telemetry event', e);
   135	      }
   136	    }
   137	  }
   138	}

   140	function parseOtlpEndpoint(
   141	  otlpEndpointSetting: string | undefined,
   142	  protocol: 'grpc' | 'http',
   143	): string | undefined {
   144	  if (!otlpEndpointSetting) {
   145	    return undefined;
   146	  }
   147	  // Trim leading/trailing quotes that might come from env variables
   148	  const trimmedEndpoint = otlpEndpointSetting.replace(/^["']|["']$/g, '');
   149	
   150	  try {
   151	    const url = new URL(trimmedEndpoint);
   152	    if (protocol === 'grpc') {
   153	      // OTLP gRPC exporters expect an endpoint in the format scheme://host:port
   154	      // The `origin` property provides this, stripping any path, query, or hash.
   155	      return url.origin;
   156	    }
   157	    // For http, use the full href.
   158	    return url.href;
   159	  } catch (error) {
   160	    diag.error('Invalid OTLP endpoint URL provided:', trimmedEndpoint, error);
   161	    return undefined;
   162	  }
   163	}

   165	export async function initializeTelemetry(
   166	  config: Config,
   167	  credentials?: JWTInput,
   168	): Promise<void> {
   169	  if (!config.getTelemetryEnabled()) {
   170	    return;
   171	  }
   172	
   173	  if (telemetryInitialized) {
   174	    if (
   175	      credentials?.client_email &&
   176	      activeTelemetryEmail &&
   177	      credentials.client_email !== activeTelemetryEmail
   178	    ) {
   179	      const message = `Telemetry credentials have changed (from ${activeTelemetryEmail} to ${credentials.client_email}), but telemetry cannot be re-initialized in this process. Please restart the CLI to use the new account for telemetry.`;
   180	      debugLogger.error(message);
   181	    }
   182	    return;
   183	  }
   184	
   185	  if (config.getTelemetryUseCollector() && config.getTelemetryUseCliAuth()) {
   186	    debugLogger.error(
   187	      'Telemetry configuration error: "useCollector" and "useCliAuth" cannot both be true. ' +
   188	        'CLI authentication is only supported with in-process exporters. ' +
   189	        'Disabling telemetry.',
   190	    );
   191	    return;
   192	  }
   193	
   194	  // If using CLI auth and no credentials provided, defer initialization
   195	  if (config.getTelemetryUseCliAuth() && !credentials) {
   196	    // Register a callback to initialize telemetry when the user logs in.
   197	    // This is done only once.
   198	    if (!callbackRegistered) {
   199	      callbackRegistered = true;
   200	      authListener = async (newCredentials: JWTInput) => {
   201	        if (config.getTelemetryEnabled() && config.getTelemetryUseCliAuth()) {
   202	          debugLogger.log('Telemetry reinit with credentials.');
   203	          await initializeTelemetry(config, newCredentials);
   204	        }
   205	      };
   206	      authEvents.on('post_auth', authListener);
   207	    }
   208	    debugLogger.log(
   209	      'CLI auth is requested but no credentials, deferring telemetry initialization.',
   210	    );
   211	    return;
   212	  }
   213	
   214	  const resource = resourceFromAttributes({
   215	    [SemanticResourceAttributes.SERVICE_NAME]: SERVICE_NAME,
   216	    [SemanticResourceAttributes.SERVICE_VERSION]: process.version,
   217	    'session.id': config.getSessionId(),
   218	  });
   219	
   220	  if (!keychainAvailabilityListener) {
   221	    keychainAvailabilityListener = (event: KeychainAvailabilityEvent) => {
   222	      logKeychainAvailability(config, event);
   223	    };
   224	    coreEvents.on(
   225	      CoreEvent.TelemetryKeychainAvailability,
   226	      keychainAvailabilityListener,
   227	    );
   228	  }
   229	
   230	  if (!tokenStorageTypeListener) {
   231	    tokenStorageTypeListener = (event: TokenStorageInitializationEvent) => {
   232	      logTokenStorageInitialization(config, event);
   233	    };
   234	    coreEvents.on(
   235	      CoreEvent.TelemetryTokenStorageType,
   236	      tokenStorageTypeListener,
   237	    );
   238	  }
   239	
   240	  const otlpEndpoint = config.getTelemetryOtlpEndpoint();
   241	  const otlpProtocol = config.getTelemetryOtlpProtocol();
   242	  const telemetryTarget = config.getTelemetryTarget();
   243	  const useCollector = config.getTelemetryUseCollector();
   244	
   245	  const parsedEndpoint = parseOtlpEndpoint(otlpEndpoint, otlpProtocol);
   246	  const telemetryOutfile = config.getTelemetryOutfile();
   247	  const useOtlp = !!parsedEndpoint && !telemetryOutfile;
   248	
   249	  const gcpProjectId =
   250	    process.env['OTLP_GOOGLE_CLOUD_PROJECT'] ||
   251	    process.env['GOOGLE_CLOUD_PROJECT'];
   252	  const useDirectGcpExport =
   253	    telemetryTarget === TelemetryTarget.GCP && !useCollector;
   254	
   255	  let spanExporter:
   256	    | OTLPTraceExporter
   257	    | OTLPTraceExporterHttp
   258	    | GcpTraceExporter
   259	    | FileSpanExporter
   260	    | ConsoleSpanExporter;
   261	  let logExporter:
   262	    | OTLPLogExporter
   263	    | OTLPLogExporterHttp
   264	    | GcpLogExporter
   265	    | FileLogExporter
   266	    | ConsoleLogRecordExporter;
   267	
   268	  if (useDirectGcpExport) {
   269	    debugLogger.log(
   270	      'Creating GCP exporters with projectId:',
   271	      gcpProjectId,
   272	      'using',
   273	      credentials ? 'provided credentials' : 'ADC',
   274	    );
   275	    spanExporter = new GcpTraceExporter(gcpProjectId, credentials);
   276	    logExporter = new GcpLogExporter(gcpProjectId, credentials);
   277	    metricReader = new PeriodicExportingMetricReader({
   278	      exporter: new GcpMetricExporter(gcpProjectId, credentials),
   279	      exportIntervalMillis: 30000,
   280	    });
   281	  } else if (useOtlp) {
   282	    if (otlpProtocol === 'http') {
   283	      const buildUrl = (path: string) => {
   284	        const url = new URL(parsedEndpoint);
   285	        // Join the existing pathname with the new path, handling trailing slashes.
   286	        url.pathname = [url.pathname.replace(/\/$/, ''), path].join('/');
   287	        return url.href;
   288	      };
   289	      spanExporter = new OTLPTraceExporterHttp({
   290	        url: buildUrl('v1/traces'),
   291	      });
   292	      logExporter = new OTLPLogExporterHttp({
   293	        url: buildUrl('v1/logs'),
   294	      });
   295	      metricReader = new PeriodicExportingMetricReader({
   296	        exporter: new OTLPMetricExporterHttp({
   297	          url: buildUrl('v1/metrics'),
   298	        }),
   299	        exportIntervalMillis: 10000,
   300	      });
